#include #include struct TriggerCalibration { uint16_t min_raw; // Rest position voltage (0% travel) uint16_t max_raw; // Fully depressed voltage (100% travel) }; TriggerCalibration leftTrigger = {1800, 3400}; // Safe defaults // Read raw ADC and convert to smooth 0.0 to 1.0 float float getCalibratedTrigger(uint8_t pin, const TriggerCalibration& cal) { uint16_t raw = analogRead(pin); // Apply deadzones (5% padding at top and bottom prevents drift) uint16_t deadzone_min = cal.min_raw + ((cal.max_raw - cal.min_raw) * 0.05); uint16_t deadzone_max = cal.max_raw - ((cal.max_raw - cal.min_raw) * 0.05); if (raw <= deadzone_min) return 0.0f; if (raw >= deadzone_max) return 1.0f; // Linear mapping between deadzones return (float)(raw - deadzone_min) / (float)(deadzone_max - deadzone_min); } // Interactive calibration procedure void runCalibrationRoutine(uint8_t pin, TriggerCalibration& cal) { Serial.println("Starting Calibration..."); Serial.println("Release trigger completely. Sampling REST position..."); delay(2000); // Sample rest position uint32_t sum = 0; for(int i = 0; i < 100; i++) { sum += analogRead(pin); delay(10); } cal.min_raw = sum / 100; Serial.println("Squeeze trigger FULLY and hold. Sampling MAX position..."); delay(3000); // Sample max position sum = 0; for(int i = 0; i < 100; i++) { sum += analogRead(pin); delay(10); } cal.max_raw = sum / 100; // Save calibration struct to onboard EEPROM/NVS Flash EEPROM.put(0, cal); EEPROM.commit(); Serial.println("Calibration complete and saved to Flash!"); }
